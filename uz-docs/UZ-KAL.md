```js
await fetch("/api/agenda.php", {method:"POST", credentials:"include",
  headers:{"Content-Type":"application/json"},
  body: JSON.stringify({action:"update", project:"urduzaban", by:"claude", light:"green",
    brief:{ur:"…", en:"…"}, next:{ur:"…", en:"…"}, note:{ur:"…", en:"…"}})}).then(r=>r.json());
```
